===============================================================================class ContextMemory:
    def __init__(self):
        """
        Bộ nhớ ngữ cảnh (Context Memory) lưu trữ trạng thái bối cảnh hiện tại 
        (ví dụ: mã nhận diện nhánh, trạng thái môi trường hiện tại) để định hướng 
        cho mạng xử lý đúng hướng khi gặp các tín hiệu lặp lại.
        """
        self.current_context = {}

    def set_context(self, key, value):
        """
        Thiết lập hoặc cập nhật một giá trị bối cảnh.
        """
        self.current_context[key] = value

    def get_context(self, key):
        """
        Lấy giá trị của một bối cảnh cụ thể.
        """
        return self.current_context.get(key, None)

    def clear(self):
        """
        Xóa toàn bộ ngữ cảnh khi đổi tác vụ hoặc kết thúc phiên.
        """
        self.current_context = {}

===============================================================================class SequenceMemory:
    def __init__(self, max_history_length=10):
        """
        Bộ nhớ chuỗi (Sequence Memory) lưu trữ lịch sử các vector trạng thái 
        hoặc token theo thứ tự thời gian để phục vụ cho việc dự đoán và phân tích chuỗi.
        """
        self.max_history_length = max_history_length
        self.history = []

    def append(self, latent_vector):
        """
        Thêm một vector mới vào lịch sử chuỗi. Giữ kích thước trong giới hạn max_history_length.
        """
        self.history.append(list(latent_vector))
        if len(self.history) > self.max_history_length:
            self.history.pop(0) # Giữ lại cửa sổ trượt lịch sử gần nhất

    def get_sequence(self):
        """
        Trả về toàn bộ chuỗi lịch sử hiện tại.
        """
        return self.history

    def reset(self):
        """
        Xóa lịch sử chuỗi khi kết thúc chuỗi hiện tại.
        """
        self.history = []

===============================================================================class TemporalTrace:
    def __init__(self, decay_rate=0.8):
        """
        Quản lý dấu vết thời gian (Temporal Trace / Eligibility Trace) 
        để lưu giữ dư âm kích thích qua các bước thời gian.
        """
        self.decay_rate = decay_rate
        self.current_trace = []

    def update(self, current_vector):
        """
        Cập nhật dấu vết thời gian: mờ dần (decay) dấu vết cũ và cộng thêm tín hiệu mới.
        Trace = Trace_cũ * decay_rate + Vector_mới
        """
        if not self.current_trace:
            self.current_trace = list(current_vector)
            return self.current_trace

        # Đảm bảo kích thước khớp nhau
        if len(self.current_trace) != len(current_vector):
            self.current_trace = list(current_vector)
            return self.current_trace

        updated = []
        for old_val, new_val in zip(self.current_trace, current_vector):
            decayed_val = (old_val * self.decay_rate) + new_val
            updated.append(round(decayed_val, 4))
            
        self.current_trace = updated
        return self.current_trace

    def reset(self):
        """
        Làm sạch dấu vết khi bắt đầu một episode hoặc chuỗi mới.
        """
        self.current_trace = []

===============================================================================class WorkingMemory:
    def __init__(self, capacity=5):
        """
        Bộ nhớ làm việc (Working Memory) đóng vai trò RAM ngắn hạn,
        giới hạn số lượng mục thông tin được giữ active trong tầm nhìn trực tiếp của mạng.
        """
        self.capacity = capacity
        self.slots = []

    def load(self, item):
        """
        Đưa một item (token/latent vector) mới vào bộ nhớ làm việc. 
        Nếu vượt quá dung lượng (capacity), item cũ nhất sẽ bị đẩy ra (FIFO).
        """
        self.slots.append(item)
        if len(self.slots) > self.capacity:
            self.slots.pop(0)

    def get_contents(self):
        """
        Trích xuất toàn bộ nội dung đang có trong RAM ngắn hạn.
        """
        return self.slots

    def clear(self):
        """
        Làm sạch bộ nhớ làm việc.
        """
        self.slots = []

class AdaptiveLIFSoma(BaseSoma):
    def __init__(self, theta=1.0, leak=0.9, a=0.01, b=0.2, tau_w=10.0, refractory_period=2):
        super().__init__(refractory_period)
        self.theta = theta
        self.leak = leak
        self.a = a
        self.b = b
        self.tau_w = tau_w
        self.w = 0.0

    def step(self, current):
        if self.refractory_left > 0:
            self.refractory_left -= 1
            return 0

        dw = (self.a * self.V - self.w) / self.tau_w
        self.w += dw
        self.V = self.V * self.leak + current - self.w

        if self.V >= self.theta:
            self.V = 0.0
            self.w += self.b
            self.refractory_left = self.refractory_period
            return 1
        return 0

from .base_soma import BaseSoma

class ELIFSoma(BaseSoma):
    def __init__(self, E_L=-65.0, V_T=-55.0, Delta_T=2.0, g_L=0.1, V_peak=0.0, refractory_period=2):
        super().__init__(refractory_period)
        self.E_L = E_L
        self.V = E_L
        self.V_T = V_T
        self.Delta_T = Delta_T
        self.g_L = g_L
        self.V_peak = V_peak

    def step(self, current):
        if self.refractory_left > 0:
            self.refractory_left -= 1
            return 0

        exp_term = self.g_L * self.Delta_T * math.exp(min((self.V - self.V_T) / self.Delta_T, 10))
        dV = -self.g_L * (self.V - self.E_L) + exp_term + current
        self.V += dV * 0.1

        if self.V >= self.V_peak:
            self.V = self.E_L
            self.refractory_left = self.refractory_period
            return 1
        return 0

from .base_soma import BaseSoma

class HodgkinHuxleySoma(BaseSoma):
    def __init__(self, E_K=-12.0, E_Na=115.0, E_L=10.613, g_K=36.0, g_Na=120.0, g_L=0.3, C_m=1.0):
        super().__init__(refractory_period=0)
        self.E_K = E_K
        self.E_Na = E_Na
        self.E_L = E_L
        self.g_K = g_K
        self.g_Na = g_Na
        self.g_L = g_L
        self.C_m = C_m

        self.V = 0.0
        self.m = 0.05
        self.h = 0.6
        self.n = 0.317
        self.was_above_threshold = False

    def step(self, current):
        alpha_m = (2.5 - 0.1 * self.V) / (math.exp(2.5 - 0.1 * self.V) - 1.0) if self.V != 25 else 1.0
        beta_m = 4.0 * math.exp(-self.V / 18.0)

        alpha_h = 0.07 * math.exp(-self.V / 20.0)
        beta_h = 1.0 / (math.exp(3.0 - 0.1 * self.V) + 1.0)

        alpha_n = (0.1 - 0.01 * self.V) / (math.exp(1.0 - 0.1 * self.V) - 1.0) if self.V != 10 else 0.1
        beta_n = 0.125 * math.exp(-self.V / 80.0)

        dt = 0.01
        for _ in range(10):
            self.m += dt * (alpha_m * (1 - self.m) - beta_m * self.m)
            self.h += dt * (alpha_h * (1 - self.h) - beta_h * self.h)
            self.n += dt * (alpha_n * (1 - self.n) - beta_n * self.n)

            I_Na = self.g_Na * (self.m ** 3) * self.h * (self.V - self.E_Na)
            I_K = self.g_K * (self.n ** 4) * (self.V - self.E_K)
            I_L = self.g_L * (self.V - self.E_L)

            dV = (current - I_Na - I_K - I_L) / self.C_m
            self.V += dV * dt

        spike = 0
        if self.V > 80.0 and not self.was_above_threshold:
            spike = 1
            self.was_above_threshold = True
        elif self.V <= 80.0:
            self.was_above_threshold = False

        return spike

class IzhikevichSoma(BaseSoma):
    def __init__(self, a=0.02, b=0.2, c=-65.0, d=8.0):
        super().__init__(refractory_period=0)
        self.a = a
        self.b = b
        self.c = c
        self.d = d
        self.V = c
        self.u = b * self.V

    def step(self, current):
        dv = (0.04 * (self.V ** 2) + 5 * self.V + 140 - self.u + current)
        du = self.a * (self.b * self.V - self.u)

        self.V += dv * 0.5
        self.u += du * 0.5

        if self.V >= 30.0:
            self.V = self.c
            self.u += self.d
            return 1
        return 0

class Neuron:

    def __init__(
        self,
        theta=1.0,
        leak=0.9,
        refractory_period=2,
        rate_alpha=0.01
    ):
        self.axon = Axon()

        self.dendrite = Dendrite()

        self.soma = Soma(
            theta=theta,
            leak=leak,
            refractory_period=refractory_period
        )

        self.rate_alpha = rate_alpha

        self.last_spike_time = -1

        self.fire_count = 0

        self.steps = 0

        self.firing_rate = 0.0

    def step(self, time):

        self.steps += 1

        current = self.dendrite.current(time)

        spike = self.soma.step(
            current
        )

        if spike:

            self.fire_count += 1

            self.last_spike_time = (
                time
            )

            self.axon.transmit(
                spike,
                time
            )

        self.firing_rate = (
            (1 - self.rate_alpha)
            * self.firing_rate
            +
            self.rate_alpha
            * spike
        )

        return spike

    def reset(self):

        self.soma.reset()

        self.last_spike_time = -1

        self.fire_count = 0

        self.steps = 0

        self.firing_rate = 0.0

class Synapse:
    def __init__(
        self,
        weight=0.5,
        delay=1,
        pre=None,
        post=None
    ):
        self.weight = weight

        self.delay = max(1, delay)

        self.queue = deque()

        self.last_output = 0.0

        self.pre_spikes = deque(
            maxlen=32
        )
        self.pre = pre
        self.post = post

        # --- Bổ sung Eligibility Trace ---
        self.eligibility_trace = 0.0
        self.trace_decay = 0.95  # Tỷ lệ suy giảm theo thời gian/step
        self.learning_rate = 0.01

    def receive(
        self,
        spike,
        time
     ):

        if spike:

            self.pre_spikes.append(time)

            self.queue.append(
                (
                    time + self.delay,
                    spike
                )
            )

    def transmit(self, current_time):
        total_spike = 0

        while (
            self.queue
            and
            self.queue[0][0] <= current_time
        ):

            _, spike = self.queue.popleft()

            total_spike += spike

        self.last_output = (
            total_spike * self.weight
        )

        return self.last_output

    def update_trace(self, pre_spiked, post_spiked):
        """Cập nhật vết dấu hiệu năng dựa trên xung của pre và post neuron"""
        self.eligibility_trace *= self.trace_decay
        if pre_spiked or post_spiked:
            self.eligibility_trace += 1.0

    def apply_reward_modulation(self, reward):
        """Áp dụng phần thưởng dựa trên eligibility trace đã tích lũy"""
        if self.eligibility_trace != 0.0:
            self.weight += self.learning_rate * reward * self.eligibility_trace
            self.clamp()

    def clamp(self):

        self.weight = max(
            0.0,
            min(5.0, self.weight)
        )

class RewardManager:

    def __init__(
        self,
        positive_reward=1.0,
        negative_reward=-1.0
    ):
        self.positive_reward = (
            positive_reward
        )

        self.negative_reward = (
            negative_reward
        )

        self.rstdp = (
            RewardModulatedSTDP()
        )

    def compute_reward(
        self,
        action,
        target
    ):

        if action == target:
            return self.positive_reward

        return self.negative_reward

    def apply_reward(
        self,
        network,
        reward
    ):
        """
        Áp dụng reward lên toàn bộ
        synapse trong network.
        """

        for synapse in network.synapses:

            post = synapse.post

            if post is None:
                continue

            if (
                post.last_spike_time < 0
            ):
                continue

            self.rstdp.update(
                synapse=synapse,
                post_spike_time=
                post.last_spike_time,
                reward=reward
            )

    def evaluate(
        self,
        network,
        action,
        target
    ):

        reward = (
            self.compute_reward(
                action,
                target
            )
        )

        self.apply_reward(
            network,
            reward
        )

        return reward

===============================================================================class SubRegion:
    def __init__(self, region_id, input_indices, output_index):
        self.region_id = region_id
        self.input_indices = input_indices
        self.output_index = output_index
        self.is_frozen = False
        self.recent_performance = []

    def update_performance(self, is_correct):
        self.recent_performance.append(1 if is_correct else 0)
        # Theo dõi qua cửa sổ 100 lần xuất hiện của task để đảm bảo độ chín muồi
        if len(self.recent_performance) > 100:
            self.recent_performance.pop(0)

    @property
    def accuracy(self):
        # Cần ít nhất 60 mẫu đánh giá thực tế mới bắt đầu xét đóng băng
        if not self.recent_performance or len(self.recent_performance) < 60:
            return 0.0
        return sum(self.recent_performance) / len(self.recent_performance) * 100.0


class SubRegionManager:
    def __init__(self, freeze_threshold=98.0):
        self.regions = {}
        self.freeze_threshold = freeze_threshold

    def register_region(self, region_id, input_indices, output_index):
        self.regions[region_id] = SubRegion(region_id, input_indices, output_index)

    def update_region_performance(self, target, is_correct):
        for region_id, region in self.regions.items():
            if region.output_index == target:
                region.update_performance(is_correct)
                if not region.is_frozen and region.accuracy >= self.freeze_threshold:
                    region.is_frozen = True
                    print(f"\n[SUB-REGION CONSOLIDATION] Tiểu vùng '{region_id}' đạt độ chính xác {region.accuracy:.1f}%. Tiến hành đóng băng cấu trúc synapse cục bộ!")

    def is_target_frozen(self, target):
        for region in self.regions.values():
            if region.output_index == target and region.is_frozen:
                return True
        return False

        for neuron in network.neurons:
            self.homeostasis.update(neuron)
            self.structural.prune(neuron)
            self.plasticity.update_neuron(neuron)

        neurons = network.neurons
        for pre in neurons:
            for post in neurons:
                if pre is post:
                    continue

                synapse = self.structural.maybe_grow(
                    pre, 
                    post, 
                    delay=1, 
                    input_neurons=self.input_neurons, 
                    output_neurons=self.output_neurons,
                    current_synapse_count=len(network.synapses)
                )
                if synapse:
                    network.add_synapse(synapse)

    def train_step(self, network, action, target):
        # Cập nhật cấu trúc mạng theo tiểu vùng
        self.update_network(network, target=target)

        # Nếu tiểu vùng đã đóng băng, khóa luôn việc thay đổi trọng số
        if self.sub_region_manager and target is not None and self.sub_region_manager.is_target_frozen(target):
            return

        if action is not None and target is not None:
            # Đánh giá phần thưởng từ RewardManager
            reward = self.reward_manager.evaluate(network, action, target)

            # Lan truyền phần thưởng thông qua Eligibility Trace của từng synapse
            for synapse in network.synapses:
                if hasattr(synapse, 'apply_reward_modulation'):
                    synapse.apply_reward_modulation(reward)

        for neuron in self.neurons:

            neuron.reset()

class RewardModulatedSTDP:

    def __init__(
        self,
        A_plus=0.005,
        A_minus=0.005,
        tau=20.0,
        reward_scale=1.0,
        weight_decay=0.0005
    ):
        self.A_plus = A_plus
        self.A_minus = A_minus

        self.tau = tau

        self.reward_scale = (
            reward_scale
        )

        self.weight_decay = (
            weight_decay
        )

    def update(
        self,
        synapse,
        post_spike_time,
        reward
    ):

        if not synapse.pre_spikes:
            return

        pre_spike_time = (
            synapse.pre_spikes[-1]
        )

        dt = (
            post_spike_time
            - pre_spike_time
        )

        stdp_term = 0.0

        if dt > 0:

            stdp_term = (
                self.A_plus
                * math.exp(
                    -dt / self.tau
                )
            )

        elif dt < 0:

            stdp_term = -(
                self.A_minus
                * math.exp(
                    dt / self.tau
                )
            )

        dw = (
            reward
            * self.reward_scale
            * stdp_term
        )

        synapse.weight += dw

        synapse.weight *= (
            1.0
            -
            self.weight_decay
        )

        synapse.clamp()

        return dw

class STDP:

    def __init__(
        self,
        A_plus=0.005,
        A_minus=0.005,
        tau=20.0,
        weight_decay=0.0005
    ):
        self.A_plus = A_plus
        self.A_minus = A_minus
        self.tau = tau

        self.weight_decay = weight_decay

    def update(
        self,
        synapse,
        post_spike_time
    ):

        if not synapse.pre_spikes:
            return

        pre_spike_time = (
            synapse.pre_spikes[-1]
        )

        dt = (
            post_spike_time
            - pre_spike_time
        )

        if dt > 0:

            dw = (
                self.A_plus
                * math.exp(
                    -dt / self.tau
                )
            )

            synapse.weight += dw

        elif dt < 0:

            dw = (
                self.A_minus
                * math.exp(
                    dt / self.tau
                )
            )

            synapse.weight -= dw

        # decay phải ở đây
        synapse.weight *= (
            1.0 - self.weight_decay
        )

        synapse.clamp()

class StructuralPlasticity:

    def __init__(
        self,
        prune_threshold=0.05,
        growth_weight=0.10,
        activity_threshold=0.60,  # Nâng ngưỡng kích hoạt lên cao để hạn chế mọc bừa bãi
        max_synapses=35           # Đặt trần giới hạn tối đa số lượng synapse trong mạng
    ):
        self.prune_threshold = prune_threshold
        self.growth_weight = growth_weight
        self.activity_threshold = activity_threshold
        self.max_synapses = max_synapses

    def prune(self, neuron):
        before = len(neuron.dendrite.synapses)
        neuron.dendrite.synapses = [
            syn for syn in neuron.dendrite.synapses
            if syn.weight > self.prune_threshold
        ]
        return before - len(neuron.dendrite.synapses)

    def grow(
        self,
        pre_neuron,
        post_neuron,
        delay=1
    ):
        return connect(
            pre_neuron,
            post_neuron,
            weight=self.growth_weight,
            delay=delay
        )

    def maybe_grow(
        self,
        pre,
        post,
        delay=1,
        input_neurons=None,
        output_neurons=None,
        current_synapse_count=0
    ):
        """
        Mọc synapse thông minh theo tiểu vùng (Sub-region / Modular Growth) 
        kèm theo kiểm soát giới hạn trần số lượng synapse.
        """
        # Nếu tổng số synapse đã đạt giới hạn trần, tuyệt đối không mọc thêm
        if current_synapse_count >= self.max_synapses:
            return None

        if pre is post:
            return None

        if self.connected(pre, post):
            return None

        # Nếu có định nghĩa danh sách tiểu vùng, giới hạn phạm vi mọc trong cụm độc lập
        if input_neurons and output_neurons:
            if pre in input_neurons and post in output_neurons:
                try:
                    idx_pre = input_neurons.index(pre)
                    idx_post = output_neurons.index(post)
                    if idx_pre != idx_post:
                        return None 
                except ValueError:
                    pass

        # Chỉ cho phép mọc khi tần số hoạt động vượt qua ngưỡng cao mới
        if (
            pre.firing_rate > self.activity_threshold and  
            post.firing_rate > self.activity_threshold
        ):
            return self.grow(pre, post, delay)

        return None

    def connected(
        self,
        pre,
        post
    ):
        for syn in pre.axon.synapses:
            if syn.post is post:
                return True
        return False

class Environment:

    def __init__(self):
        self.branch_a = [0, 1, 2, 1, 0]
        self.branch_b = [0, 1, 0, 2, 0]
        
        self.current_branch = self.branch_a
        self.target_sequence = self.current_branch
        self.current_step_idx = 0
        self.branch_id = 0 # 0 cho Branch A, 1 cho Branch B
        
        # Mở rộng vector trạng thái lên 9 chiều: 
        # 3 chiều đầu (giá trị), 5 chiều sau (vị trí bước), 1 chiều cuối (Branch Flag)
        self.current_state_vector = [1, 0, 0, 1, 0, 0, 0, 0, 1]

    def reset(self):
        self.branch_id = random.choice([0, 1])
        self.current_branch = self.branch_a if self.branch_id == 0 else self.branch_b
        self.target_sequence = self.current_branch
        self.current_step_idx = 0
        self.update_state_vector()
        return self.current_state_vector, self.target_sequence[self.current_step_idx]

    def update_state_vector(self):
        target = self.target_sequence[self.current_step_idx]

        # 1. Mã hóa giá trị target (3 chiều đầu)
        val_vec = [0, 0, 0]
        if target == 0:
            val_vec = [1, 0, 0]
        elif target == 1:
            val_vec = [0, 1, 0]
        elif target == 2:
            val_vec = [0, 0, 1]

        # 2. Mã hóa vị trí bước (5 chiều tiếp theo - One-hot encoding)
        pos_vec = [0, 0, 0, 0, 0]
        if 0 <= self.current_step_idx < len(self.target_sequence):
            pos_vec[self.current_step_idx] = 1

        # 3. Mã hóa Branch Flag (1 chiều cuối: [1, 0] cho nhánh A, [0, 1] hoặc giá trị phân biệt cho nhánh B)
        # Ở đây ta dùng 1 chiều đơn giản: 1 nếu là nhánh A, 0 nếu là nhánh B
        branch_vec = [1 if self.branch_id == 0 else 0]

        self.current_state_vector = val_vec + pos_vec + branch_vec

    def observe(self):
        target = self.target_sequence[self.current_step_idx]
        return self.current_state_vector, target

    def step(self, action):
        expected_target = self.target_sequence[self.current_step_idx]

        if action == expected_target:
            reward = 1.0
            self.current_step_idx += 1

            if self.current_step_idx >= len(self.target_sequence):
                done = True
                self.current_step_idx = 0
            else:
                done = False

            self.update_state_vector()
        else:
            reward = -1.0
            done = True
            self.current_step_idx = 0
            self.update_state_vector()

        return self.current_state_vector, reward, done

    def reward(self, action, target):
        if action == target:
            return 1.0
        return -1.0

    def evaluate(self, action, target):
        reward = self.reward(action, target)
        done = reward > 0
        return reward, done

===============================================================================class EpisodeRunner:

    def __init__(
        self,
        environment,
        network,
        input_neurons,
        agent,
        trainer,
        reward_manager,
    ):
        self.environment = environment
        self.network = network
        self.input_neurons = input_neurons
        self.agent = agent
        self.trainer = trainer
        self.reward_manager = reward_manager

    def run_episode(self, steps_per_action=30, learning=True):
        state_vector, target = self.environment.reset()
        done = False
        total_reward = 0.0
        steps_count = 0
        last_outputs = []
        action = None

        max_steps = len(getattr(self.environment, 'target_sequence', [0, 1, 2]))

        while not done and steps_count < max_steps:
            steps_count += 1

            for _ in range(max(0, steps_per_action)):
                # Kích hoạt tất cả các input neuron có giá trị 1 trong vector 8 chiều
                for i, val in enumerate(state_vector):
                    if val == 1 and i < len(self.input_neurons):
                        self.input_neurons[i].stimulate(
                            spike=1,
                            time=self.network.clock.time,
                        )

                last_outputs = self.network.step()

            action = self.agent.act()
            next_state_vector, reward, done = self.environment.step(action)
            total_reward += reward

            is_step_correct = (reward > 0)

            if learning:
                self.trainer.train_step(self.network, action, target)
                if hasattr(self.trainer, 'sub_region_manager') and self.trainer.sub_region_manager:
                    self.trainer.sub_region_manager.update_region_performance(target, is_step_correct)

            target = getattr(self.environment, 'target_sequence', [target])[self.environment.current_step_idx] if hasattr(self.environment, 'current_step_idx') else target
            state_vector = next_state_vector

        is_correct = (total_reward >= len(getattr(self.environment, 'target_sequence', [0, 1, 2])))

        return {
            "pattern": state_vector,
            "target": target,
            "action": action,
            "reward": total_reward,
            "correct": is_correct,
            "outputs": last_outputs,
            "synapses": len(self.network.synapses),
        }

    def train(self, episodes=150):
        history = []
        recent_correct = []

        for episode in range(max(0, episodes)):
            result = self.run_episode(steps_per_action=30, learning=True)
            history.append(result)
            recent_correct.append(1 if result["correct"] else 0)

            if len(recent_correct) > 20:
                recent_correct.pop(0)

            if episode % 10 == 0:
                acc = sum(recent_correct) / len(recent_correct) * 100
                print(
                    f"episode={episode:02d} | "
                    f"target={result['target']} | "
                    f"action={result['action']} | "
                    f"reward={result['reward']:+.1f} | "
                    f"acc_20={acc:.1f}% | "
                    f"syn={result['synapses']}"
                )

        return history

    def evaluate(self, num_tests=30):
        correct = 0
        print("\n--- BẮT ĐẦU KIỂM THỬ TUẦN TỰ (EVALUATION MODE) ---")
        for _ in range(num_tests):
            result = self.run_episode(steps_per_action=30, learning=False)
            if result["correct"]:
                correct += 1
            print(f"Test Chuỗi -> Target: {result['target']} | Agent Action: {result['action']} ({'ĐÚNG' if result['correct'] else 'SAI'})")

        accuracy = (correct / num_tests) * 100
        print(f"Độ chính xác tổng kết chuỗi: {accuracy:.1f}% ({correct}/{num_tests})\n")
        return accuracy

    @staticmethod
    def estimate(prediction_error, max_error_bound=2.0):
        """

class PredictionError:
    @staticmethod
    def calculate(predicted_vector, actual_vector):
        """
        Tính toán khoảng cách sai lệch (Sai số dự đoán) giữa trạng thái dự đoán 
        và trạng thái thực tế xảy ra trong môi trường.
        """
        if not predicted_vector or not actual_vector or len(predicted_vector) != len(actual_vector):
            return 1.0 # Sai số tối đa nếu thiếu dữ liệu hoặc lệch chiều
            
        # Tính sai số bình phương trung bình (MSE) hoặc Euclidean distance
        sum_sq_diff = sum((p - a) ** 2 for p, a in zip(predicted_vector, actual_vector))
        mse = sum_sq_diff / len(predicted_vector)
        
        return round(math.sqrt(mse), 4) # Trả về giá trị sai số chuẩn hóa

        """

        để truy hồi và so khớp độ tương đồng giữa các khái niệm/trạng thái.
        """
        self.memory_space = {} # Lưu dạng { "label_name": [latent_vector] }

    def register_feature(self, key, latent_vector):
        """
        Đăng ký hoặc cập nhật một vector đặc trưng cho một khóa (nhãn/khái niệm).
        """
        self.memory_space[key] = latent_vector

    def find_closest(self, input_vector):
        """
        Tìm khóa (key) có latent vector gần nhất với input_vector dựa trên Cosine Similarity.
        Giúp hệ thống nhận diện vật thể/trạng thái tương tự.
        """
        if not self.memory_space:
            return None, 0.0

        best_match = None
        highest_similarity = -1.0

        for key, stored_vector in self.memory_space.items():
            similarity = LatentVector.cosine_similarity(input_vector, stored_vector)
            if similarity > highest_similarity:
                highest_similarity = similarity
                best_match = key

        return best_match, highest_similarity

class LatentVector:
    @staticmethod
    def cosine_similarity(vec_a, vec_b):
        """
        Tính độ tương đồng góc giữa hai latent vector (giúp hệ thống nhận biết 
        'vật này giống vật kia' dựa trên góc không gian ẩn).
        """
        if not vec_a or not vec_b or len(vec_a) != len(vec_b):
            return 0.0
            
        dot_product = sum(a * b for a, b in zip(vec_a, vec_b))
        magnitude_a = math.sqrt(sum(a ** 2 for a in vec_a))
        magnitude_b = math.sqrt(sum(b ** 2 for b in vec_b))
        
        if magnitude_a == 0.0 or magnitude_b == 0.0:
            return 0.0
            
        return dot_product / (magnitude_a * magnitude_b)

    @staticmethod
    def euclidean_distance(vec_a, vec_b):
        """
        Tính khoảng cách Euclidean giữa hai latent vector trong không gian đặc trưng.
        """
        if not vec_a or not vec_b or len(vec_a) != len(vec_b):
            return float('inf')
            
        sum_sq_diff = sum((a - b) ** 2 for a, b in zip(vec_a, vec_b))
        return math.sqrt(sum_sq_diff)

class PopulationCoder:
    def __init__(self, num_neurons=10, min_val=0.0, max_val=2.0, sigma=0.5):
        """
        Mã hóa quần thể (Population Coding) chuyển đổi một giá trị scalar thô
        thành một phân phối kích thích trên một quần thể nơ-ron.
        """
        self.num_neurons = num_neurons
        self.min_val = min_val
        self.max_val = max_val
        self.sigma = sigma
        
        # Chia đều các tâm tiếp nhận (preferred values) cho các nơ-ron trong quần thể
        self.centers = [
            min_val + i * (max_val - min_val) / (num_neurons - 1)
            if num_neurons > 1 else min_val
            for i in range(num_neurons)
        ]

    def encode(self, value):
        """
        Nhận vào một giá trị (ví dụ: nhãn 0, 1, 2 hoặc một giá trị liên tục)
        và trả về vector kích thích quần thể (Latent Vector dạng float từ 0.0 đến 1.0).
        """
        encoded_vector = []
        for center in self.centers:
            # Hàm Gaussian tuning curve tính độ kích hoạt của nơ-ron theo khoảng cách tới tâm
            activation = math.exp(-((value - center) ** 2) / (2 * (self.sigma ** 2)))
            encoded_vector.append(round(activation, 4))
        return encoded_vector
